cmake_minimum_required(VERSION 3.21)
project(meteor_steamvr_trial LANGUAGES C CXX)
set(OPENVR_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../meteor/target/windows-openvr-driver-src" CACHE PATH "Pinned Valve OpenVR SDK")
add_library(driver_meteor SHARED driver.cpp transport.c)
target_compile_features(driver_meteor PRIVATE cxx_std_17)
target_compile_features(driver_meteor PRIVATE c_std_11)
target_compile_definitions(driver_meteor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(driver_meteor PRIVATE "${OPENVR_SDK}/headers")
set_target_properties(driver_meteor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/meteor/bin/win64>")
add_custom_command(TARGET driver_meteor POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/driver.vrdrivermanifest" "${CMAKE_BINARY_DIR}/meteor/driver.vrdrivermanifest")
find_package(Vulkan REQUIRED)
set(NV_CODEC_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../meteor/target/windows-vibepollo-src/third-party/nv-codec-headers/include/ffnvcodec" CACHE PATH "Pinned NVENC headers")
target_include_directories(driver_meteor PRIVATE "${NV_CODEC_HEADERS}")
target_link_libraries(driver_meteor PRIVATE d3d11 dxgi d3dcompiler Vulkan::Vulkan ws2_32)
add_executable(meteor-steamvr-mirror mirror.cpp transport.c)
target_compile_features(meteor-steamvr-mirror PRIVATE cxx_std_17)
target_compile_features(meteor-steamvr-mirror PRIVATE c_std_11)
target_compile_definitions(meteor-steamvr-mirror PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(meteor-steamvr-mirror PRIVATE "${OPENVR_SDK}/headers" "${NV_CODEC_HEADERS}")
target_link_libraries(meteor-steamvr-mirror PRIVATE d3d11 dxgi d3dcompiler Vulkan::Vulkan ws2_32)
add_executable(meteor-steamvr-geometry-test geometry_test.cpp)
target_compile_features(meteor-steamvr-geometry-test PRIVATE cxx_std_17)
target_compile_options(meteor-steamvr-geometry-test PRIVATE /UNDEBUG)
target_include_directories(meteor-steamvr-geometry-test PRIVATE "${OPENVR_SDK}/headers")
